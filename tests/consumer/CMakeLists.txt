cmake_minimum_required(VERSION 3.21)

set(ARIXV_CONSUMER_LANG "CXX" CACHE STRING "C or CXX")
if(ARIXV_CONSUMER_LANG STREQUAL "C")
  project(arixv_consumer_smoke LANGUAGES C)
elseif(ARIXV_CONSUMER_LANG STREQUAL "CXX")
  project(arixv_consumer_smoke LANGUAGES CXX)
else()
  message(FATAL_ERROR "Unknown ARIXV_CONSUMER_LANG: ${ARIXV_CONSUMER_LANG}")
endif()

set(ARIXV_CONSUMER_MODE "find_package" CACHE STRING "find_package or fetchcontent")
set(ARIXV_CONSUMER_SOURCE_DIR "" CACHE PATH "Path to arix-v source dir for fetchcontent mode")
option(ARIXV_CONSUMER_LINK_ENCODER "Link encoder target (legacy switch; overridden by ARIXV_CONSUMER_TARGETS if set)" ON)
set(ARIXV_CONSUMER_TARGETS "" CACHE STRING "both, decoder, or encoder (empty derives from ARIXV_CONSUMER_LINK_ENCODER)")

if(ARIXV_CONSUMER_TARGETS STREQUAL "")
  if(ARIXV_CONSUMER_LINK_ENCODER)
    set(_targets_mode "both")
  else()
    set(_targets_mode "decoder")
  endif()
else()
  set(_targets_mode "${ARIXV_CONSUMER_TARGETS}")
endif()

if(ARIXV_CONSUMER_MODE STREQUAL "fetchcontent")
  include(FetchContent)
  if(_targets_mode STREQUAL "decoder")
    set(ARIXV_BUILD_ENCODER OFF CACHE BOOL "" FORCE)
  else()
    set(ARIXV_BUILD_ENCODER ON CACHE BOOL "" FORCE)
  endif()
  FetchContent_Declare(arixv
    SOURCE_DIR "${ARIXV_CONSUMER_SOURCE_DIR}"
  )
  FetchContent_MakeAvailable(arixv)
elseif(ARIXV_CONSUMER_MODE STREQUAL "find_package")
  find_package(arixv REQUIRED)
else()
  message(FATAL_ERROR "Unknown ARIXV_CONSUMER_MODE: ${ARIXV_CONSUMER_MODE}")
endif()

if(ARIXV_CONSUMER_LANG STREQUAL "C")
  add_executable(consumer_smoke main.c)
  set_target_properties(consumer_smoke PROPERTIES
    C_STANDARD 11
    C_STANDARD_REQUIRED ON
    C_EXTENSIONS OFF
  )
  if(MSVC)
    target_compile_options(consumer_smoke PRIVATE /W4 /WX)
  else()
    target_compile_options(consumer_smoke PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
else()
  add_executable(consumer_smoke main.cpp)
  target_compile_features(consumer_smoke PRIVATE cxx_std_17)
  set_target_properties(consumer_smoke PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(consumer_smoke PRIVATE /W4 /WX)
  else()
    target_compile_options(consumer_smoke PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()

if(_targets_mode STREQUAL "both")
  # Intentionally link BOTH targets into one executable to verify diamond static
  # builds and single-DSO shared builds on ELF, Mach-O, and Windows DLLs.
  target_link_libraries(consumer_smoke PRIVATE arixv::decoder arixv::encoder)
  target_compile_definitions(consumer_smoke PRIVATE CONSUMER_HAS_ENCODER=1)
elseif(_targets_mode STREQUAL "encoder")
  target_link_libraries(consumer_smoke PRIVATE arixv::encoder)
  target_compile_definitions(consumer_smoke PRIVATE CONSUMER_HAS_ENCODER=1)
elseif(_targets_mode STREQUAL "decoder")
  target_link_libraries(consumer_smoke PRIVATE arixv::decoder)
  target_compile_definitions(consumer_smoke PRIVATE CONSUMER_HAS_ENCODER=0)
else()
  message(FATAL_ERROR "Unknown ARIXV_CONSUMER_TARGETS: ${_targets_mode}")
endif()
